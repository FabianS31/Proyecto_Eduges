-- ============================================================
-- EduGes - Base de Datos
-- PostgreSQL
-- ============================================================

BEGIN;

-- ============================================================
-- 1. TABLAS DE CATÁLOGO
-- ============================================================

CREATE TABLE IF NOT EXISTS Parentescos (
    ID_Parentesco SERIAL PRIMARY KEY,
    Descripcion VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS Tipos_Obras_Sociales (
    ID_TipoOS SERIAL PRIMARY KEY,
    Tipo VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS Estados_Pacientes (
    ID_EstadoPaciente SERIAL PRIMARY KEY,
    Estado VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS Estados_Turnos (
    ID_EstadoTurno SERIAL PRIMARY KEY,
    Estado VARCHAR(50) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS Especialidades (
    ID_Especialidad SERIAL PRIMARY KEY,
    Especialidad VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS Roles (
    ID_Rol SERIAL PRIMARY KEY,
    Rol VARCHAR(50) NOT NULL UNIQUE
);

-- ============================================================
-- 2. OBRAS SOCIALES
-- ============================================================

CREATE TABLE IF NOT EXISTS Obras_Sociales (
    ID_ObraSocial SERIAL PRIMARY KEY,
    Nombre VARCHAR(150) NOT NULL,
    Contacto VARCHAR(150),
    Mail VARCHAR(150),
    Web VARCHAR(255),
    ID_TipoOS INTEGER NOT NULL,

    CONSTRAINT FK_ObrasSociales_Tipos
        FOREIGN KEY (ID_TipoOS)
        REFERENCES Tipos_Obras_Sociales(ID_TipoOS)
);

-- ============================================================
-- 3. TUTORES
-- ============================================================

CREATE TABLE IF NOT EXISTS Tutores (
    ID_Tutor SERIAL PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Apellido VARCHAR(100) NOT NULL,
    Telefono VARCHAR(30),
    Movil VARCHAR(30),
    Mail VARCHAR(150)
);

-- ============================================================
-- 4. PROFESIONALES
-- ============================================================

CREATE TABLE IF NOT EXISTS Profesionales (
    ID_Profesional SERIAL PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Apellido VARCHAR(100) NOT NULL,
    DNI VARCHAR(20) NOT NULL UNIQUE,
    Matricula VARCHAR(50) NOT NULL UNIQUE,
    ID_Especialidad INTEGER NOT NULL,
    ID_Rol INTEGER NOT NULL,

    CONSTRAINT FK_Profesionales_Especialidades
        FOREIGN KEY (ID_Especialidad)
        REFERENCES Especialidades(ID_Especialidad),

    CONSTRAINT FK_Profesionales_Roles
        FOREIGN KEY (ID_Rol)
        REFERENCES Roles(ID_Rol)
);

-- ============================================================
-- 5. PACIENTES
-- ============================================================

CREATE TABLE IF NOT EXISTS Pacientes (
    ID_Paciente SERIAL PRIMARY KEY,
    Nombre VARCHAR(100) NOT NULL,
    Apellido VARCHAR(100) NOT NULL,
    DNI VARCHAR(20) NOT NULL UNIQUE,
    FechaNacimiento DATE NOT NULL,
    Direccion VARCHAR(255),
    Telefono VARCHAR(30),
    CUD_Vencimiento DATE NOT NULL,
    ID_ObraSocial INTEGER NOT NULL,
    ID_EstadoPaciente INTEGER NOT NULL,

    CONSTRAINT FK_Pacientes_ObrasSociales
        FOREIGN KEY (ID_ObraSocial)
        REFERENCES Obras_Sociales(ID_ObraSocial),

    CONSTRAINT FK_Pacientes_Estados
        FOREIGN KEY (ID_EstadoPaciente)
        REFERENCES Estados_Pacientes(ID_EstadoPaciente)
);

-- ============================================================
-- 6. RELACIÓN PACIENTES - TUTORES
-- ============================================================

CREATE TABLE IF NOT EXISTS Pacientes_Tutores (
    ID_PacienteTutor SERIAL PRIMARY KEY,
    ID_Tutor INTEGER NOT NULL,
    ID_Paciente INTEGER NOT NULL,
    ID_Parentesco INTEGER NOT NULL,
    ResponsablePrincipal BOOLEAN NOT NULL DEFAULT FALSE,

    CONSTRAINT FK_PacientesTutores_Tutor
        FOREIGN KEY (ID_Tutor)
        REFERENCES Tutores(ID_Tutor)
        ON DELETE CASCADE,

    CONSTRAINT FK_PacientesTutores_Paciente
        FOREIGN KEY (ID_Paciente)
        REFERENCES Pacientes(ID_Paciente)
        ON DELETE CASCADE,

    CONSTRAINT FK_PacientesTutores_Parentesco
        FOREIGN KEY (ID_Parentesco)
        REFERENCES Parentescos(ID_Parentesco),

    CONSTRAINT UQ_Paciente_Tutor
        UNIQUE (ID_Paciente, ID_Tutor)
);

-- ============================================================
-- 7. UN ÚNICO RESPONSABLE PRINCIPAL POR PACIENTE
-- ============================================================

CREATE UNIQUE INDEX IF NOT EXISTS UQ_Paciente_ResponsablePrincipal
ON Pacientes_Tutores (ID_Paciente)
WHERE ResponsablePrincipal = TRUE;

-- ============================================================
-- 8. REGISTROS DE SESIONES
-- ============================================================

CREATE TABLE IF NOT EXISTS Registros_De_Sesiones (
    ID_RegistroSesion SERIAL PRIMARY KEY,
    TratamientoAplicado VARCHAR(5000)
);

-- ============================================================
-- 9. ASIGNACIONES DE PROFESIONALES
-- ============================================================

CREATE TABLE IF NOT EXISTS Asignaciones_Profesionales (
    ID_AsignacionProfesional SERIAL PRIMARY KEY,
    ID_Paciente INTEGER NOT NULL,
    ID_Profesional INTEGER NOT NULL,
    FechaInicio DATE NOT NULL,
    FechaFin DATE,

    CONSTRAINT FK_AsignacionesProfesionales_Paciente
        FOREIGN KEY (ID_Paciente)
        REFERENCES Pacientes(ID_Paciente),

    CONSTRAINT FK_AsignacionesProfesionales_Profesional
        FOREIGN KEY (ID_Profesional)
        REFERENCES Profesionales(ID_Profesional),

    CONSTRAINT CK_AsignacionesProfesionales_Fechas
        CHECK (
            FechaFin IS NULL
            OR FechaFin >= FechaInicio
        )
);

-- ============================================================
-- 10. DÍAS DE LA SEMANA
-- ============================================================

CREATE TABLE IF NOT EXISTS Dias_Semana (
    ID_Dia SERIAL PRIMARY KEY,
    Dia VARCHAR(20) NOT NULL UNIQUE
);

-- ============================================================
-- 11. PROGRAMACIONES DE TURNOS
-- ============================================================

CREATE TABLE IF NOT EXISTS Programaciones_Turnos (
    ID_ProgramacionTurno SERIAL PRIMARY KEY,
    ID_Paciente INTEGER NOT NULL,
    ID_Profesional INTEGER NOT NULL,
    FechaInicio DATE NOT NULL,
    FechaFin DATE NOT NULL,

    CONSTRAINT FK_ProgramacionesTurnos_Paciente
        FOREIGN KEY (ID_Paciente)
        REFERENCES Pacientes(ID_Paciente),

    CONSTRAINT FK_ProgramacionesTurnos_Profesional
        FOREIGN KEY (ID_Profesional)
        REFERENCES Profesionales(ID_Profesional),

    CONSTRAINT CK_ProgramacionesTurnos_Fechas
        CHECK (FechaFin >= FechaInicio)
);

-- ============================================================
-- 12. DÍAS Y HORARIOS DE LAS PROGRAMACIONES
-- ============================================================

CREATE TABLE IF NOT EXISTS Programaciones_Turnos_Dias (
    ID_ProgramacionDia SERIAL PRIMARY KEY,
    ID_ProgramacionTurno INTEGER NOT NULL,
    ID_Dia INTEGER NOT NULL,
    HoraInicio TIME NOT NULL,
    HoraFin TIME NOT NULL,

    CONSTRAINT FK_ProgramacionesTurnosDias_Programacion
        FOREIGN KEY (ID_ProgramacionTurno)
        REFERENCES Programaciones_Turnos(ID_ProgramacionTurno)
        ON DELETE CASCADE,

    CONSTRAINT FK_ProgramacionesTurnosDias_Dia
        FOREIGN KEY (ID_Dia)
        REFERENCES Dias_Semana(ID_Dia),

    CONSTRAINT UQ_Programacion_Dia
        UNIQUE (ID_ProgramacionTurno, ID_Dia),

    CONSTRAINT CK_ProgramacionesTurnosDias_Horario
        CHECK (HoraFin > HoraInicio)
);

-- ============================================================
-- 13. TURNOS
-- ============================================================

CREATE TABLE IF NOT EXISTS Turnos (
    ID_Turno SERIAL PRIMARY KEY,
    Fecha DATE NOT NULL,
    Hora TIME NOT NULL,
    ID_EstadoTurno INTEGER NOT NULL,
    ID_RegistroSesion INTEGER,
    ID_Profesional INTEGER NOT NULL,
    ID_Paciente INTEGER NOT NULL,

    CONSTRAINT FK_Turnos_Estados
        FOREIGN KEY (ID_EstadoTurno)
        REFERENCES Estados_Turnos(ID_EstadoTurno),

    CONSTRAINT FK_Turnos_RegistroSesion
        FOREIGN KEY (ID_RegistroSesion)
        REFERENCES Registros_De_Sesiones(ID_RegistroSesion)
        ON DELETE SET NULL,

    CONSTRAINT FK_Turnos_Profesional
        FOREIGN KEY (ID_Profesional)
        REFERENCES Profesionales(ID_Profesional),

    CONSTRAINT FK_Turnos_Paciente
        FOREIGN KEY (ID_Paciente)
        REFERENCES Pacientes(ID_Paciente)
);

-- ============================================================
-- 14. DATOS INICIALES DE CATÁLOGOS
-- ============================================================

INSERT INTO Estados_Pacientes (Estado)
VALUES
    ('Activo'),
    ('Inactivo')
ON CONFLICT (Estado) DO NOTHING;

INSERT INTO Estados_Turnos (Estado)
VALUES
    ('Pendiente'),
    ('Confirmado'),
    ('Cancelado'),
    ('Realizado')
ON CONFLICT (Estado) DO NOTHING;

INSERT INTO Roles (Rol)
VALUES
    ('Administrador'),
    ('Profesional')
ON CONFLICT (Rol) DO NOTHING;

INSERT INTO Dias_Semana (Dia)
VALUES
    ('Lunes'),
    ('Martes'),
    ('Miércoles'),
    ('Jueves'),
    ('Viernes'),
    ('Sábado'),
    ('Domingo')
ON CONFLICT (Dia) DO NOTHING;

COMMIT;
